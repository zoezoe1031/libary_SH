구글 스프레드시트 연동 코드 (Apps Script)

1. 구글 스프레드시트 생성 > [확장 프로그램] > [Apps Script] 클릭
2. 기존 코드 전체를 지우고 아래 코드를 붙여넣기 후 [배포] > [새 배포] 클릭
3. 액세스 권한을 '모든 사용자(Anyone)'로 설정하여 배포 후 URL을 복사하세요. 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 헤더 자동 생성
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["제출시각", "구분", "학번/부서", "이름", "독서날짜", "책제목", "페이지수", "마라톤거리(km)", "독서일지"]);
    }
    
    sheet.appendRow([
      new Date(),
      data.userType,
      data.info1,
      data.userName,
      data.readDate,
      data.bookTitle,
      data.pageCount,
      data.pageCount, // 1페이지 = 1km
      data.journalText
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
